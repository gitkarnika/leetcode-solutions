## Solution

```cpp
class Solution {
public:
    string minWindow(string s, string t) {
        vector<int> tele(256,0),sele(256,0);
        int having=0,need=0,minlen=INT_MAX,bestind,left=0;
        for(int i=0;i<t.size();i++){
            tele[t[i]]++;
        }
        for(int i=0;i<256;i++){
            if(tele[i]>=1) need++;
        }
        for(int right=0;right<s.size();right++){
            sele[s[right]]++;
            if(sele[s[right]]==tele[s[right]]) having++;
            while(having==need){
                if(right-left+1<minlen){
                    bestind=left;
                    minlen=min(minlen,right-left+1);
                }
                sele[s[left]]--;
                if(sele[s[left]]<tele[s[left]]) having--;
                left++;
            }
        }
        if(minlen==INT_MAX) return "";
        return s.substr(bestind,minlen);
    }
};
```
