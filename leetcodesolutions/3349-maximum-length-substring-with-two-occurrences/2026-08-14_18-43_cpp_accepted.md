## Solution

```cpp
class Solution {
public:
    int maximumLengthSubstring(string s) {
        unordered_map<int,int> freq;
        int maxlen=0,left=0;
        for(int right=0;right<s.size();right++){
            freq[s[right]]++;
            while(freq[s[right]]>2){
                freq[s[left]]--;
                left++;
            }
            maxlen=max(maxlen,right-left+1);
        }
        return maxlen;
    }
};
```
