## Solution

```cpp
class Solution {
public:
    vector<int> findMissingElements(vector<int>& nums) {
        vector<int> ans;
        sort(nums.begin(),nums.end());
        for(int i=0;i<nums.size()-1;i++){
            int p=nums[i+1]-nums[i];
            while(p>1){
                nums[i]=nums[i]+1;
                ans.push_back(nums[i]);
                p--;
            }
        }
        return ans;
    }
};
```
