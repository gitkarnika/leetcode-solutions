## Solution

```cpp
class Solution {
public:
    void rotate(vector<vector<int>>& matrix) {
        int temp;
        for(int i=0;i<matrix.size();++i){
            for(int j=i+1;j<matrix.size();++j){
                temp=matrix[i][j];
                matrix[i][j]=matrix[j][i];
                matrix[j][i]=temp;
            }
        }
        for(int i=0;i<matrix.size();++i){
            reverse(matrix[i].begin(),matrix[i].end());
        }
    }
};
```
