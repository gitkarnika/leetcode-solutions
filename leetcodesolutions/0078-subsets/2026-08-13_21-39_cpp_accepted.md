## Solution

```cpp
class Solution {
public:
    vector<vector<int>> subsets(vector<int>& nums) {
        int pow=1<<nums.size();
        vector<vector<int>> fans;
        for(int i=0;i<pow;i++){
            vector<int> ans;
            for(int j=0;j<nums.size();j++){
                if(i&(1<<j)) ans.push_back(nums[j]);
            }
            fans.push_back(ans);
        }
        return fans;
    }
};
```
