## Solution

```cpp
class Solution {
public:
    int maxProduct(vector<int>& nums) {
        long long maxend=nums[0],minend=nums[0],ans=nums[0];
        for(int i=1;i<nums.size();i++){
            long long x=nums[i];
            long long candmax=max({x,maxend*x,minend*x});
            long long candmin=min({x,maxend*x,minend*x});
            maxend=candmax;
            minend=candmin;
            ans=max(ans,maxend);
        }
        return ans;
    }
};
```
