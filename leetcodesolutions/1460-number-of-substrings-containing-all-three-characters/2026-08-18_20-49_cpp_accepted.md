## Solution

```cpp
class Solution {
public:
    int numberOfSubstrings(string s) {
        int count=0,left=0;
        vector<int> freq(3,0);
        for(int i=0;i<s.size();i++){
            freq[s[i]-'a']++;
            while(freq[0]>0 && freq[1]>0 && freq[2]>0){
                count+=(s.length()-i);
                freq[s[left]-'a']--;
                left++;
            }
        }
        return count;
    }
};
```
