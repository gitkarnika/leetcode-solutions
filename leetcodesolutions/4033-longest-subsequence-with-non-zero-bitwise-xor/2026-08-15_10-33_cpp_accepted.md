## Solution

```cpp
class Solution {
public:
    int longestSubsequence(vector<int>& nums) {
        int XOR=0;
        long long sum=0;
        for(int right=0;right<nums.size();right++){
            XOR^=nums[right];
            sum+=nums[right];
        }
        if(XOR!=0) return nums.size();
        if(sum==0) return 0;
        return nums.size()-1;
    }
};
```
